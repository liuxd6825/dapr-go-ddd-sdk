# 提示词上下文管理模块 - 实现规划

## 一、模块定位

在 `pkg/ai/rag/my_rag/context/` 目录下新增上下文管理模块，负责：
- 动态组装 Prompt（System Prompt + 历史对话 + RAG上下文 + 当前查询）
- Token 精确预算与安全截断
- 上下文分级优先级调度
- 可观测性日志记录

## 二、核心架构设计

```
context/
├── context_manager.go    # 主入口，负责整体上下文组装管线
├── priority.go           # 上下文分级策略（T0-T3）
├── token_budget.go        # Token 预算计算与分配
├── truncator.go          # 安全截断（按轮次截断）
├── compressor.go         # 记忆压缩（滑动摘要、实体提取）

├── reranker.go           # 上下文重排
├── template.go           # Jinja2 模板引擎封装
├── logger.go             # 上下文日志记录（可观测性）

```

```go
package context

type Priority int // T0=0, T1=1, T2=2, T3=3

const (
    T0_SystemPrompt      Priority = 0  // 绝对保留，放两头
    T0_UserQuery        Priority = 0  // 当前用户问题，放最后

    Content     string
    TokenSize   int
    Metadata    map[string]any // 存储来源、角色等元数据
    ID          string         // 用于日志追踪
}

// 组装顺序：T0(头) → T3 → T2 → T1 → T0(尾)
func (p Priority) IsHead() bool { return p == T0_SystemPrompt }
func (p Priority) IsTail() bool { return p == T0_UserQuery }
```

## 四、核心组件设计

### 4.1 Token 计算器

```go
// context/token_budget.go
package context

import (
    "github.com/pkoukk/tiktoken-go"
)

type TokenBudget struct {
    encoding     *tiktoken.Tiktoken
    maxTokens    int
    reserved     int
    available    int // 剩余可用
}

func NewTokenBudget(maxTokens, reserved int) (*TokenBudget, error) {
    encoding, err := tiktoken.GetEncoding("cl100k_base")
    if err != nil {
        return nil, err
    }
    return &TokenBudget{
        encoding:  encoding,
        maxTokens: maxTokens,
        reserved:  reserved,
        available: maxTokens - reserved,
    }, nil
}

// CountTokens 计算文本对应的 token 数量
func (tb *TokenBudget) CountTokens(text string) int {
    return len(tb.encoding.Encode(text, nil, nil))
}

// Allocate 按比例分配 token 预算
func (tb *TokenBudget) Allocate(ratio float64) int {
    return int(float64(tb.available) * ratio)
}

// Remaining 返回剩余可用 token
func (tb *TokenBudget) Remaining() int {
    return tb.available
}

// Consume 消耗 token
func (tb *TokenBudget) Consume(tokens int) {
    tb.available -= tokens
    if tb.available < 0 {
        tb.available = 0
    }
}
```

```go
// context/priority.go
package context

type PriorityManager struct {
    items map[Priority][]*ContextItem
}

func NewPriorityManager() *PriorityManager {
    return &PriorityManager{
        items: make(map[Priority][]*ContextItem),
    }
}

// Add 添加上下文项
func (pm *PriorityManager) Add(item *ContextItem) {
    pm.items[item.Priority] = append(pm.items[item.Priority], item)
}

// GetByPriority 获取指定优先级的所有项
func (pm *PriorityManager) GetByPriority(p Priority) []*ContextItem {
    return pm.items[p]
}

// SortForAssembly 排序为组装顺序
// 组装顺序: T0_SystemPrompt → T3 → T2 → T1 → T0_UserQuery
func (pm *PriorityManager) SortForAssembly() []*ContextItem {
    var result []*ContextItem

    // T0 System Prompt 放最前
    result = append(result, pm.items[T0_SystemPrompt]...)

    // T3 远期摘要
    result = append(result, pm.items[T3_LongTermHistory]...)

    // T2 近期对话
    result = append(result, pm.items[T2_RecentHistory]...)

    // T1 RAG 召回知识
    result = append(result, pm.items[T1_RetrievedKnowledge]...)

    // T0 User Query 放最后
    result = append(result, pm.items[T0_UserQuery]...)

    return result
}
```

### 4.3 安全截断器

```go
// context/truncator.go
package context

type Truncator struct {
    budget      *TokenBudget
    maxTurns    int
}

func NewTruncator(budget *TokenBudget, maxTurns int) *Truncator {
    return &Truncator{
        budget:   budget,
        maxTurns: maxTurns,
    }
}

// TruncateConversationHistory 按对话轮次安全截断
// 截断单位：一整轮（User + Assistant），不从句子中间切断
func (t *Truncator) TruncateConversationHistory(history []*Content) []*Content {
    if len(history) == 0 {
        return history
    }

    var result []*Content
    var currentTurn []*Content
    remaining := t.budget.Remaining()

    // 从最新开始保留
    for i := len(history) - 1; i >= 0; i-- {
        item := history[i]
        tokens := t.budget.CountTokens(item.Content)

        if remaining < tokens && len(currentTurn) > 0 {
            break // 不再添加更早的内容
        }

        currentTurn = append([]*Content{item}, currentTurn...)
        remaining -= tokens

        // 遇到 user 消息认为是一轮结束
        if item.Role == "user" {
            if len(result)+len(currentTurn) > t.maxTurns*2 {
                break
            }
            result = append(currentTurn, result...)
            currentTurn = nil
        }
    }

    // 反转回正序
    for i, j := 0, len(result)-1; i < j; i, j = i+1, j-1 {
        result[i], result[j] = result[j], result[i]
    }

    return result
}
```

### 4.5 查询重写器

```go
// context/query_rewriter.go
package context

type QueryRewriter struct {
    llm llm.LLM
}

func NewQueryRewriter(llm llm.LLM) *QueryRewriter {
    return &QueryRewriter{llm: llm}
}

// Rewrite 将多轮对话中的指代消解，重写为独立完整的查询
func (qr *QueryRewriter) Rewrite(ctx context.Context, currentQuery string, history []*Content) (string, error) {
    if len(history) == 0 {
        return currentQuery, nil
    }

    historyText := ""
    for _, h := range history {
        historyText += fmt.Sprintf("%s: %s\n", h.Role, h.Content)
    }

    prompt := fmt.Sprintf(`根据对话历史，将当前问题改写为一个独立、完整的查询。

对话历史：
%s

当前问题：%s

改写要求：
1. 将指代词（如"它"、"这个"）替换为具体内容
2. 补全省略的主语和宾语
3. 确保问题可以独立理解

只输出改写后的查询语句，不要其他内容：`, historyText, currentQuery)

    msg := []*schema.Message{{Role: schema.User, Content: prompt}}
    resp, err := qr.llm.Generate(ctx, msg)
    if err != nil {
        return currentQuery, err
    }

    return strings.TrimSpace(resp.Content), nil
}
```

### 4.6 上下文重排器

```go
// context/reranker.go
package context

type Reranker struct {
    // 可接入 BGE-Reranker 等重排模型
    rerankModel string
}

func NewReranker(rerankModel string) *Reranker {
    return &Reranker{rerankModel: rerankModel}
}

// Reorder 根据与 query 的相关性重排 chunks
// 最相关的放最接近用户问题的位置（最末端），次相关的放最前端
func (r *Reranker) Reorder(ctx context.Context, query string, chunks []string) ([]string, error) {
    if len(chunks) <= 1 {
        return chunks, nil
    }

    // 1. 调用 rerank 模型获取相关性分数
    scores, err := r.callRerankModel(ctx, query, chunks)
    if err != nil {
        return chunks, err // 降级为原始顺序
    }

    // 2. 按分数排序
    sorted := make([]struct{ chunk string; score float64 }, len(chunks))
    for i, chunk := range chunks {
        sorted[i] = struct{ chunk string; score float64 }{chunk, scores[i]}
    }
    sort.Slice(sorted, func(i, j int) bool {
        return sorted[i].score > sorted[j].score
    })

    // 3. 重排：次相关放前端，最相关放尾端
    result := make([]string, 0, len(chunks))
    for i := len(sorted) - 1; i >= 0; i-- {
        result = append(result, sorted[i].chunk)
    }

    return result, nil
}
```

### 4.7 Jinja2 模板封装

```go
// context/template.go
package context

import (
    "strings"
    "text/template"
)

var defaultPromptTemplate = `{{.SystemPrompt}}

{{if .Summary}}
【历史摘要】：{{.Summary}}
{{end}}

{{if .History}}
【历史对话】：
{{- range .History}}
{{.Role}}: {{.Content}}
{{- end}}
{{end}}

{{if .RetrievedContext}}
【参考知识】：
{{- range $i, .RetrievedContext}}
[{{$i}}] {{.}}
{{- end}}
{{end}}

{{.CurrentQuery}}

回答：`

type TemplateData struct {
    SystemPrompt    string
    Summary         string
    History         []*Content
    RetrievedContext []string
    CurrentQuery    string
}

func RenderPrompt(data *TemplateData) (string, error) {
    tmpl := template.New("rag-prompt").Funcs(template.FuncMap{
        "add": func(a, b int) int { return a + b },
    })

    var err error
    tmpl, err = tmpl.Parse(defaultPromptTemplate)
    if err != nil {
        return "", err
    }

    buf := &strings.Builder{}
    if err := tmpl.Execute(buf, data); err != nil {
        return "", err
    }

    return buf.String(), nil
}
```

### 4.8 主上下文管理器

```go
// context/context_manager.go
package context

import (
    "context"
)

type ContextManager struct {
    config    *Config
    budget    *TokenBudget
    priority  *PriorityManager
    truncator *Truncator
    compressor *Compressor
    rewriter  *QueryRewriter
    reranker  *Reranker
    logger    *ContextLogger
}

func NewContextManager(config *Config) (*ContextManager, error) {
    budget, err := NewTokenBudget(config.MaxTotalTokens, config.ReservedTokens)
    if err != nil {
        return nil, err
    }

    return &ContextManager{
        config:    config,
        budget:    budget,
        priority:  NewPriorityManager(),
        truncator: NewTruncator(budget, config.MaxHistoryTurns),
        compressor: NewCompressor(nil, SummaryConfig{
            Threshold: config.SummaryThreshold,
        }),
        rewriter: nil,
        reranker: NewReranker(""),
        logger:   NewContextLogger(),
    }
}

// Build 组装完整上下文
func (cm *ContextManager) Build(ctx context.Context, param *BuildParam) (*BuildResult, error) {
    cm.priority = NewPriorityManager() // 重置

    // 1. 添加 System Prompt（T0）
    cm.priority.Add(&ContextItem{
        Priority:  T0_SystemPrompt,
        Content:   cm.config.SystemPrompt,
        TokenSize: cm.budget.CountTokens(cm.config.SystemPrompt),
        ID:        "system",
    })

    // 2. 查询重写（如果启用）
    query := param.Query
    if cm.config.EnableQueryRewrite {
        rewritten, err := cm.rewriter.Rewrite(ctx, query, param.ConversationHistory)
        if err == nil {
            query = rewritten
        }
    }

    // 3. 处理对话历史
    var summary string
    if cm.config.EnableSummary {
        historyTokens := cm.budget.CountTokens(serializeHistory(param.ConversationHistory))
        if historyTokens > cm.config.SummaryThreshold {
            summary, _ = cm.compressor.GenerateSummary(ctx, param.ConversationHistory)
            cm.priority.Add(&ContextItem{
                Priority:  T3_LongTermHistory,
                Content:   summary,
                TokenSize: cm.budget.CountTokens(summary),
                ID:        "summary",
            })
        }
    }

    // 4. 截断并添加近期对话（T2）
    truncatedHistory := cm.truncator.TruncateConversationHistory(param.ConversationHistory)
    for _, h := range truncatedHistory {
        cm.priority.Add(&ContextItem{
            Priority:  T2_RecentHistory,
            Content:   h.Content,
            TokenSize: cm.budget.CountTokens(h.Content),
            ID:        "history-" + h.Role,
        })
    }

    // 5. 处理 RAG 召回上下文（T1）
    ragBudget := cm.budget.Allocate(cm.config.RAGRatio)
    chunks := param.RetrievedContext

    if cm.config.EnableReranker && len(chunks) > 1 {
        chunks, _ = cm.reranker.Reorder(ctx, query, chunks)
    }

    var selectedChunks []string
    var usedTokens int
    for _, chunk := range chunks {
        chunkTokens := cm.budget.CountTokens(chunk)
        if usedTokens+chunkTokens > ragBudget {
            break
        }
        selectedChunks = append(selectedChunks, chunk)
        usedTokens += chunkTokens
    }

    for i, chunk := range selectedChunks {
        cm.priority.Add(&ContextItem{
            Priority:  T1_RetrievedKnowledge,
            Content:   chunk,
            TokenSize: cm.budget.CountTokens(chunk),
            ID:        "rag-chunk-" + strconv.Itoa(i),
        })
    }

    // 6. 添加当前查询（T0）
    cm.priority.Add(&ContextItem{
        Priority:  T0_UserQuery,
        Content:   query,
        TokenSize: cm.budget.CountTokens(query),
        ID:        "user-query",
    })

    // 7. 组装最终 Prompt
    items := cm.priority.SortForAssembly()
    promptData := &TemplateData{
        SystemPrompt:    cm.config.SystemPrompt,
        Summary:         summary,
        History:         truncatedHistory,
        RetrievedContext: selectedChunks,
        CurrentQuery:    query,
    }
    finalPrompt, _ := RenderPrompt(promptData)

    // 8. 记录日志
    result := &BuildResult{
        FinalPrompt:    finalPrompt,
        UsedTokens:     cm.budget.CountTokens(finalPrompt),
        ContextItems:   items,
        QueryRewritten: query,
    }
    cm.logger.Log(result)

    return result, nil
}

type BuildParam struct {
    Query             string
    RetrievedContext  []string
    ConversationHistory []*Content
}
```

```go
// graph_rag.go 修改
func (g *GraphRag) Query(ctx context.Context, query *QueryParam, streams ...func(txt string)) (string, error) {
    if query.Query == "" {
        return "", errors.New("query is empty")
    }

    // 构建上下文管理器
    ctxManager, err := context.NewContextManager(&context.Config{

    })

    }

    // 获取图数据上下文
    re := g.getGraphContext(ctx, query)
    if re.Err != nil {
        return "", re.Err
    }

    // 构建上下文
    buildParam := &context.BuildParam{
        Query:              query.Query,
        RetrievedContext:   re.Data,
        ConversationHistory: query.ConversationHistory,
    }

    ctxResult, err := ctxManager.Build(ctx, buildParam)
    if err != nil {
        return "", err
    }

    // 使用组装的 prompt 调用 LLM

    }

    sb, err := Reader(resp, streams...)

        return "", fmt.Errorf("读取返回结果时出错: %w", err)
```
